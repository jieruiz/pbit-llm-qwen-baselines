cmake_minimum_required(VERSION 3.25)
project(pbit_dramsys_bridge LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(DRAMSYS_BUILD_CLI OFF CACHE BOOL "" FORCE)
set(DRAMSYS_BUILD_TRACE_ANALYZER OFF CACHE BOOL "" FORCE)
set(DRAMSYS_USE_FETCH_CONTENT ON CACHE BOOL "" FORCE)
set(DRAMSYS_USE_DRAMPOWER ON CACHE BOOL "" FORCE)
if(NOT DRAMSYS_SOURCE)
  include(FetchContent)
  FetchContent_Declare(dramsys_source
    GIT_REPOSITORY https://github.com/tukl-msd/DRAMSys.git
    GIT_TAG 9f15aaf1fe6f05ff16a2e521a20a96f3bfe29d52)
  FetchContent_GetProperties(dramsys_source)
  if(NOT dramsys_source_POPULATED)
    FetchContent_Populate(dramsys_source)
  endif()
  set(DRAMSYS_SOURCE ${dramsys_source_SOURCE_DIR})
endif()
add_subdirectory(${DRAMSYS_SOURCE} upstream)
add_executable(pbit_dram_bridge bridge.cpp)
target_link_libraries(pbit_dram_bridge PRIVATE DRAMSys::DRAMSys nlohmann_json::nlohmann_json)
